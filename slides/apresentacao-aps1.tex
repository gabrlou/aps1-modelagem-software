\section{Visão Geral e o Problema da Clínica}
% =========================================================

% ---------------------------------------------------------
% Slide 3: O Problema da Clinica e o que o Sistema Resolve
% ---------------------------------------------------------
\setLayoutClean
\begin{frame}{O Problema da Clínica e o que o Sistema Resolve}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{Problemas da rotina manual (Relatório, Seção 1.2):}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item \textbf{Recepção sobrecarregada:} secretários perdem tempo atendendo ligações para marcar e desmarcar.
            \item \textbf{Marcações duplas:} sem uma agenda centralizada, dois pacientes acabam marcados no mesmo horário.
            \item \textbf{Alto número de faltas:} pacientes esquecem da consulta por falta de lembretes práticos.
            \item \textbf{Perda de histórico:} controles em papel dificultam remanejar horários e não deixam rastros para auditoria.
        \end{itemize}
        }

        \column{0.48\textwidth}
        \textbf{Como o sistema resolve na prática:}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item \textbf{Base única de horários:} grade médica dividida em blocos de 30 minutos com consulta em tempo real.
            \item \textbf{Reserva com trava atômica:} no momento da confirmação, o slot é travado no banco na hora.
            \item \textbf{Avisos automáticos:} confirmação imediata e lembrete 24 h antes por e-mail e WhatsApp em paralelo.
            \item \textbf{Auditoria imutável:} registra autor, data/hora, IP e motivo obrigatório para cancelar ou remarcar (LGPD).
        \end{itemize}
        }
    \end{columns}
\end{frame}

\setLayoutClean
\begin{frame}{Quem Usa o Sistema: Os 5 Atores e suas Responsabilidades}
    \begin{columns}[T]
        \column{0.50\textwidth}
        \textbf{Atores humanos (Relatório, Seção 1.3):}
        {\scriptsize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item \textbf{Paciente:} faz autoatendimento pelo navegador, escolhe médico e horário, agenda sozinho e pode cancelar ou remarcar com pelo menos 2 h de antecedência.
            \item \textbf{Médico:} cadastra sua disponibilidade semanal em slots de 30 min e marca o atendimento (início, fim ou falta). Não agenda nem cancela consultas.
            \item \textbf{Secretário:} acompanha as agendas, confirma consultas pendentes e pode agendar ou desmarcar para pacientes que ligam na recepção.
            \item \textbf{Administrador:} cadastra médicos, secretários e ativa especialidades. Não opera consultas.
        \end{itemize}
        }

        \column{0.48\textwidth}
        \textbf{Ator secundário:}
        {\scriptsize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item \textbf{Sistema de Notificação:} componente em segundo plano que dispara e-mails e mensagens de WhatsApp sem travar o uso da aplicação.
        \end{itemize}
        }
        \vspace{0.15cm}
        \textbf{Regra de ouro dos perfis:}
        {\scriptsize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Cada usuário tem exatamente um perfil exclusivo no sistema (\texttt{Perfil}), garantindo que o médico atenda pacientes mas não altere agendas alheias.
        \end{itemize}
        }
    \end{columns}
\end{frame}

\begin{frame}{Como o Sistema Foi Dividido: Os 6 Módulos da Aplicação}
    \begin{columns}[T]
        \column{0.50\textwidth}
        \textbf{Módulos de Acesso e Grade (Relatório, Seção 1.4):}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{2.5pt}
            \item \textbf{Cadastro e Autenticação:} exige e-mail único para todos, CPF para pacientes e CRM/UF para médicos. Senha com hash criptográfico e bloqueio após erros seguidos.
            \item \textbf{Consulta e Especialidades:} catálogo público para o paciente buscar especialidades ativas e médicos credenciados.
            \item \textbf{Agenda do Médico:} organiza a grade em blocos padrão de 30 min (\texttt{LIVRE}, \texttt{BLOQUEADO} ou \texttt{OCUPADO}) entre 30 min e 90 dias no futuro.
        \end{itemize}
        }

        \column{0.48\textwidth}
        \textbf{Módulos de Operação e Mensagens:}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{2.5pt}
            \item \textbf{Agendamento:} guia o paciente em 4 passos, faz o lock atômico da vaga e trata cancelamentos e remarcações com motivo obrigatório.
            \item \textbf{Confirmação (Recepção):} painel para o secretário contatar o paciente e registrar confirmação (opcional: médico atende mesmo sem isso).
            \item \textbf{Notificação e Auditoria:} fila de envio com retentativas automáticas e log imutável de todas as ações críticas.
        \end{itemize}
        }
    \end{columns}
\end{frame}

            \item \textbf{Liberação imediata:} no cancelamento, o slot volta na hora para \texttt{LIVRE}; na remarcação, a antiga é cancelada e cria-se uma nova.

        }

        \textbf{Prevenção de Concorrência (RN-08 e RN-10):}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{1.5pt}

            \item O sistema exibe o resumo; ao confirmar, aplica \textbf{lock atômico} no banco: ocupa o slot naquele milissegundo.
            \item Se a sessão expirou enquanto ele pensava, exige login mantendo os dados selecionados na tela.

        {\footnotesize

            \item \textbf{Falha ao gravar no banco (Fluxo E2):} faz rollback total; o horário volta a ficar \texttt{LIVRE} e avisa o erro na tela.

\begin{frame}{Comprovante e Mensagens: Por que o Envio é em Paralelo}
    \begin{columns}[T]
        \column{0.50\textwidth}
        \textbf{O problema de rede que precisava resolver:}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Provedores de e-mail e APIs de WhatsApp podem oscilar ou demorar segundos. O paciente não pode ficar com a tela travada esperando isso.
        \end{itemize}
        }
        \vspace{0.1cm}
        \textbf{Como o sistema resolve com bifurcação (fork/join):}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Ao gravar a consulta, o fluxo se divide em duas partes que andam ao mesmo tempo:
            \item \textbf{Ramo 1:} O paciente vê o comprovante com o código da consulta na hora na tela.
            \item \textbf{Ramo 2:} O \texttt{Sistema de Notificação} cuida do envio em segundo plano.
        \end{itemize}
        }

        \column{0.48\textwidth}
        \textbf{Envio resiliente por canal (Fluxo A3, RN-25 a RN-28):}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item E-mail e WhatsApp disparam em paralelo; cada canal tenta até 3 vezes com tempo de espera (\textit{backoff}).
            \item Se o WhatsApp falhar após 3 tentativas, ele fica marcado como \texttt{FALHOU}, mas o e-mail continua e a consulta permanece \texttt{AGENDADA}.
        \end{itemize}
        }
        \vspace{0.1cm}
        \textbf{Evidência no diagrama:}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Barras de \textit{fork/join} e laços com cláusula \texttt{backward:Aguardar}.
        \end{itemize}
        }
    \end{columns}
\end{frame}

\setLayoutClean
\begin{frame}{A Hierarquia de Perfis: Por que Usamos Herança}
    \begin{columns}[T]
        \column{0.50\textwidth}
        \textbf{A classe abstrata `Usuario' e atributos protegidos (\#):}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Ninguém no sistema é só um usuário genérico; por isso ela é abstrata e não pode ser criada diretamente.
            \item Centraliza o que todos têm: \texttt{id}, \texttt{nome}, \texttt{email} único, \texttt{senhaHash}, \texttt{telefone} e \texttt{status}.
            \item Os atributos usam \texttt{\#} (protegidos) para que as subclasses herdem e usem direto, sem expor métodos públicos perigosos.
        \end{itemize}
        }

        \column{0.48\textwidth}
        \textbf{O que cada perfil acrescenta de concreto (Relatório, Seção 4.1):}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item \texttt{Paciente}: adiciona \texttt{cpf} único e métodos para \texttt{agendar()}, \texttt{cancelar()} e \texttt{remarcar()}.
            \item \texttt{Medico}: adiciona \texttt{crm} e \texttt{ufCrm}, e métodos para \texttt{definirDisponibilidade()} e atender consultas.
            \item \texttt{Secretario}: adiciona \texttt{matricula} e métodos para confirmar ou agendar por telefone.
            \item \texttt{Administrador}: métodos exclusivos para cadastrar médicos, secretários e especialidades.

        {\scriptsize

        }

        \column{0.48\textwidth}
        \textbf{Relação Horário $\to$ Consulta (1 para 0..*):}
        {\scriptsize
        \begin{itemize}\setlength{\itemsep}{2pt}

            \item \textbf{Autoassociação \texttt{remarcadaDe} (0..1 para 0..1):} a nova consulta aponta para a consulta cancelada anterior.
            \item \textbf{Rastreabilidade:} associações \texttt{confirmadaPor} e \texttt{canceladaPor} com \texttt{Usuario}.
        \end{itemize}
        }
